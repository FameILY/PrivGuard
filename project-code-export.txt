Audio redaction module with smart interval merging
"""
from pydub import AudioSegment
from pydub.generators import Sine
import logging
from typing import List, Dict

logger = logging.getLogger(__name__)


class AudioRedactor:
    """Handles audio redaction with beep generation and interval merging"""
    
    def __init__(
        self,
        beep_frequency: int = 1000,
        min_beep_duration: int = 300,
        merge_gap: int = 30,
        use_silence: bool = False
    ):
        """
        Initialize audio redactor
        
        Args:
            beep_frequency: Beep frequency in Hz
            min_beep_duration: Minimum beep duration in ms
            merge_gap: Gap in ms to merge close intervals
            use_silence: Use silence instead of beep
        """
        self.beep_frequency = beep_frequency
        self.min_beep_duration = min_beep_duration
        self.merge_gap = merge_gap
        self.use_silence = use_silence
        
        logger.info(f"Audio redactor initialized:")
        logger.info(f"  Beep frequency: {beep_frequency} Hz")
        logger.info(f"  Min duration: {min_beep_duration} ms")
        logger.info(f"  Merge gap: {merge_gap} ms")
        logger.info(f"  Mode: {'SILENCE' if use_silence else 'BEEP'}")
    
    def redact_audio(self, audio_path: str, pii_segments: List[Dict]) -> AudioSegment:
        """
        Redact audio by replacing PII with beeps or silence
        
        Args:
            audio_path: Path to input audio
            pii_segments: List of PII segments to redact
        
        Returns:
            Redacted AudioSegment
        """
        logger.info(f"Loading audio: {audio_path}")
        audio = AudioSegment.from_file(audio_path)
        
        if not pii_segments:
            logger.info("No PII to redact, returning original audio")
            return audio
        
        # Convert to mono for consistency
        audio = audio.set_channels(1)
        sample_rate = audio.frame_rate
        
        # Merge close intervals
        merged_intervals = self._merge_intervals(pii_segments)
        logger.info(f"Merged into {len(merged_intervals)} intervals")
        
        # Generate base beep sound
        beep_base = self._generate_beep(
            self.min_beep_duration,
            sample_rate,
            audio.sample_width
        )
        
        # Build redacted audio
        redacted_audio = AudioSegment.empty()
        last_end = 0
        
        for i, interval in enumerate(merged_intervals):
            start_ms = int(interval['start'] * 1000)
            end_ms = int(interval['end'] * 1000)
            
            # Clip to audio bounds
            start_ms = max(0, min(start_ms, len(audio)))
            end_ms = max(0, min(end_ms, len(audio)))
            
            # Add audio before this segment
            if start_ms > last_end:
                redacted_audio += audio[last_end:start_ms]
            
            # Add beep or silence
            duration_ms = end_ms - start_ms
            if duration_ms > 0:
                if self.use_silence:
                    redacted_segment = AudioSegment.silent(duration=duration_ms)
                else:
                    redacted_segment = beep_base[:duration_ms]
                
                # Add fade to prevent clicks
                fade_duration = min(10, duration_ms // 4)
                if fade_duration > 0:
                    redacted_segment = redacted_segment.fade_in(fade_duration).fade_out(fade_duration)
                
                redacted_audio += redacted_segment
            
            last_end = end_ms
            
            if (i + 1) % 10 == 0:
                logger.info(f"Processed {i + 1}/{len(merged_intervals)} redactions")
        
        # Add remaining audio
        if last_end < len(audio):
            redacted_audio += audio[last_end:]
        
        logger.info("✅ Audio redaction complete")
        return redacted_audio
    
    def _merge_intervals(self, segments: List[Dict]) -> List[Dict]:
        """
        Merge PII segments that are close together
        
        Args:
            segments: List of PII segments
        
        Returns:
            Merged list of segments
        """
        if not segments:
            return []
        
        # Convert to milliseconds for merging
        intervals = []
        for seg in segments:
            start_ms = int(seg['start'] * 1000)
            end_ms = int(seg['end'] * 1000)
            intervals.append({
                'start': seg['start'],
                'end': seg['end'],
                'start_ms': start_ms,
                'end_ms': end_ms,
                'text': seg.get('text', ''),
                'type': seg.get('entity_type', '')
            })
        
        # Sort by start time
        intervals.sort(key=lambda x: x['start_ms'])
        
        # Merge close intervals
        merged = []
        current = intervals[0]
        
        for interval in intervals[1:]:
            # Check if close enough to merge
            if interval['start_ms'] <= current['end_ms'] + self.merge_gap:
                # Merge by extending current
                current['end'] = max(current['end'], interval['end'])
                current['end_ms'] = max(current['end_ms'], interval['end_ms'])
                current['text'] += f" + {interval['text']}"
            else:
                # Save current and start new
                merged.append(current)
                current = interval
        
        # Add last interval
        merged.append(current)
        
        return merged
    
    def _generate_beep(
        self,
        duration_ms: int,
        sample_rate: int,
        sample_width: int
    ) -> AudioSegment:
        """
        Generate a beep sound
        
        Args:
            duration_ms: Duration in milliseconds
            sample_rate: Sample rate in Hz
            sample_width: Sample width in bytes
        
        Returns:
            AudioSegment with beep
        """
        # Generate sine wave
        beep = Sine(self.beep_frequency).to_audio_segment(
            duration=duration_ms,
            volume=-20.0  # Slightly reduced volume
        )
        
        # Match audio properties
        beep = beep.set_frame_rate(sample_rate)
        beep = beep.set_sample_width(sample_width)
        
        return beep
    
    def save_audio(self, audio: AudioSegment, output_path: str, format: str = None):
        """
        Save audio to file
        
        Args:
            audio: AudioSegment to save
            output_path: Output path
            format: Output format (auto-detect if None)
        """
        if format is None:
            format = output_path.split('.')[-1]
        
        logger.info(f"Saving audio to: {output_path} (format: {format})")
        audio.export(output_path, format=format)
        logger.info("✅ Audio saved successfully")

"""
Audio Redaction Pro - Main Entry Point
Combines WhisperX transcription with multi-layer PII detection
"""
import argparse
import json
import os
import sys
from pathlib import Path

from transcriber import WhisperXTranscriber
from pii_detector import MultiLayerPIIDetector
from audio_processor import AudioRedactor
from utils import setup_logging, validate_audio_file, format_time


DEFAULT_BANNED_WORDS = {
    "fuck", "shit", "bitch", "asshole", "nigger", "slut", "dick",
    "bastard", "idiot", "stupid", "fag", "damn", "ass",
    "fucking", "stupidbitch", "fuckingasshole"
}


def main():
    parser = argparse.ArgumentParser(
        description='Audio Redaction Pro - Advanced PII redaction with WhisperX',
        formatter_class=argparse.RawDescriptionHelpFormatter,
        epilog="""
Examples:
  %(prog)s -i input/audio.wav -o output/redacted.wav
  %(prog)s -i input/call.mp3 -o output/clean.mp3 --model medium --verbose
  %(prog)s -i audio.wav -o out.wav --pii-types PERSON,PHONE --use-silence
        """
    )
    
    # Required arguments
    parser.add_argument(
        '--input', '-i',
        required=True,
        help='Path to input audio file'
    )
    parser.add_argument(
        '--output', '-o',
        required=True,
        help='Path to output redacted audio file'
    )
    
    # Model configuration
    parser.add_argument(
        '--model', '-m',
        default='medium',
        choices=['tiny', 'base', 'small', 'medium', 'large-v2'],
        help='WhisperX model size (default: medium)'
    )
    parser.add_argument(
        '--language', '-l',
        default=None,
        help='Language code (en, es, fr, etc). Auto-detect if not specified.'
    )
    
    # PII detection configuration
    parser.add_argument(
        '--banned-words',
        default=None,
        help='Comma-separated list of banned words to redact'
    )
    parser.add_argument(
        '--banned-words-file',
        default=None,
        help='Path to file containing banned words (one per line)'
    )
    parser.add_argument(
        '--pii-types',
        default='all',
        help='Comma-separated PII types to redact (default: all). Options: PERSON,ORG,GPE,PHONE,EMAIL,SSN,CREDIT_CARD'
    )
    parser.add_argument(
        '--ner-model',
        default='urchade/gliner_multi_pii-v1',
        help='HuggingFace NER model to use (default: urchade/gliner_multi_pii-v1)'
    )
    
    # Audio processing configuration
    parser.add_argument(
        '--beep-frequency',
        type=int,
        default=1000,
        help='Beep frequency in Hz (default: 1000)'
    )
    parser.add_argument(
        '--min-beep-duration',
        type=int,
        default=300,
        help='Minimum beep duration in milliseconds (default: 300)'
    )
    parser.add_argument(
        '--merge-gap',
        type=int,
        default=30,
        help='Gap in ms to merge close intervals (default: 30)'
    )
    parser.add_argument(
        '--use-silence',
        action='store_true',
        help='Use silence instead of beep for redaction'
    )
    
    # Output options
    parser.add_argument(
        '--save-transcript',
        action='store_true',
        help='Save the full transcript with timestamps'
    )
    parser.add_argument(
        '--verbose', '-v',
        action='store_true',
        help='Verbose output with detailed progress'
    )
    
    args = parser.parse_args()
    
    # Setup logging
    logger = setup_logging(verbose=args.verbose)
    
    try:
        # Validate input file
        logger.info(f"🔍 Validating input file: {args.input}")
        if not validate_audio_file(args.input):
            logger.error(f"❌ Invalid or unsupported audio file: {args.input}")
            sys.exit(1)
        
        # Create output directory if needed
        output_dir = os.path.dirname(args.output)
        if output_dir:
            os.makedirs(output_dir, exist_ok=True)
        
        # Load banned words
        banned_words = DEFAULT_BANNED_WORDS.copy()
        if args.banned_words:
            custom_words = {w.strip().lower() for w in args.banned_words.split(',')}
            banned_words.update(custom_words)
            logger.info(f"Added {len(custom_words)} custom banned words")
        
        if args.banned_words_file and os.path.exists(args.banned_words_file):
            with open(args.banned_words_file, 'r') as f:
                file_words = {line.strip().lower() for line in f if line.strip()}
                banned_words.update(file_words)
                logger.info(f"Loaded {len(file_words)} banned words from file")
        
        # Parse PII types
        pii_types = None if args.pii_types == 'all' else args.pii_types.split(',')
        
        # Step 1: Transcribe audio with WhisperX
        logger.info(f"🎤 Transcribing audio with WhisperX model '{args.model}'...")
        transcriber = WhisperXTranscriber(model_size=args.model)
        transcript_result = transcriber.transcribe(
            args.input,
            language=args.language
        )
        
        logger.info(f"✅ Transcription complete!")
        logger.info(f"   Language: {transcript_result['language']}")
        logger.info(f"   Words: {len(transcript_result['words'])}")
        logger.info(f"   Segments: {len(transcript_result['segments'])}")
        
        # Save transcript if requested
        if args.save_transcript:
            transcript_path = args.output.replace('.wav', '_transcript.json').replace('.mp3', '_transcript.json')
            with open(transcript_path, 'w', encoding='utf-8') as f:
                json.dump(transcript_result, f, indent=2, ensure_ascii=False)
            logger.info(f"💾 Transcript saved to: {transcript_path}")
        
        # Step 2: Detect PII using multi-layer approach
        logger.info("🔍 Detecting PII in transcript...")
        detector = MultiLayerPIIDetector(
            banned_words=banned_words,
            pii_types=pii_types,
            ner_model=args.ner_model
        )
        
        pii_segments = detector.detect_pii(transcript_result)
        logger.info(f"🎯 Found {len(pii_segments)} PII instances to redact")
        
        # Show detected PII summary
        if args.verbose and pii_segments:
            logger.info("\\n📋 Detected PII Preview:")
            type_counts = {}
            for seg in pii_segments:
                type_counts[seg['entity_type']] = type_counts.get(seg['entity_type'], 0) + 1
            
            for pii_type, count in sorted(type_counts.items()):
                logger.info(f"   {pii_type}: {count} instances")
            
            logger.info("\\n   First 5 detections:")
            for segment in pii_segments[:5]:
                logger.info(
                    f"   [{format_time(segment['start'])} - {format_time(segment['end'])}] "
                    f"{segment['entity_type']}: {segment['text']}"
                )
            if len(pii_segments) > 5:
                logger.info(f"   ... and {len(pii_segments) - 5} more\\n")
        
        # Step 3: Process audio and apply redaction
        logger.info("🔊 Processing audio and applying redactions...")
        processor = AudioRedactor(
            beep_frequency=args.beep_frequency,
            min_beep_duration=args.min_beep_duration,
            merge_gap=args.merge_gap,
            use_silence=args.use_silence
        )
        
        redacted_audio = processor.redact_audio(
            args.input,
            pii_segments
        )
        
        # Save redacted audio
        output_format = Path(args.output).suffix[1:]  # Remove the dot
        processor.save_audio(redacted_audio, args.output, format=output_format)
        logger.info(f"💾 Redacted audio saved to: {args.output}")
        
        # Save redaction report
        report_path = args.output.replace('.wav', '_report.json').replace('.mp3', '_report.json')
        report = {
            'input_file': args.input,
            'output_file': args.output,
            'model': args.model,
            'language': transcript_result['language'],
            'total_words': len(transcript_result['words']),
            'total_pii_instances': len(pii_segments),
            'pii_summary': {},
            'redacted_segments': pii_segments
        }
        
        # Add PII summary
        for seg in pii_segments:
            pii_type = seg['entity_type']
            if pii_type not in report['pii_summary']:
                report['pii_summary'][pii_type] = 0
            report['pii_summary'][pii_type] += 1
        
        with open(report_path, 'w', encoding='utf-8') as f:
            json.dump(report, f, indent=2, ensure_ascii=False)
        logger.info(f"📊 Redaction report saved to: {report_path}")
        
        # Final summary
        logger.info("\\n" + "="*60)
